`;

const edit = document.getElementById("edit");
const view = document.getElementById("view");
const msg  = document.getElementById("msg");

/* ===================== SEGMENTS ===================== */

function getSegments(doc) {
  const html = doc
    .replace(//gi, "")
    .replace(//gi, "");

  const js = (doc.match(/]*>([\s\S]*?)<\/script>/i) || ["",""])[1];
  const css = (doc.match(/]*>([\s\S]*?)<\/style>/i) || ["",""])[1];

  return { html, js, css };
}

function setSegment(doc, mode, content) {
  if (mode === "html") {
    return doc.replace(
      /(.*?)(|)/is,
      content + "$2"
    );
  }

  if (mode === "js") {
    return doc.replace(
      /(]*>)[\s\S]*?(<\/script>)/i,
      `$1\n${content}\n$2`
    );
  }

  if (mode === "css") {
    return doc.replace(
      /(]*>)[\s\S]*?(<\/style>)/i,
      `$1\n${content}\n$2`
    );
  }
}

/* ===================== BLOCK DETECTION ===================== */

function findBraceBlock(text, pos) {
  let start = pos, depth = 0;
  for (let i = pos; i >= 0; i--) {
    if (text[i] === "}") depth++;
    if (text[i] === "{") {
      if (depth === 0) { start = i; break; }
      depth--;
    }
  }

  depth = 0;
  let end = pos;
  for (let i = pos; i < text.length; i++) {
    if (text[i] === "{") depth++;
    if (text[i] === "}") {
      if (depth === 0) { end = i + 1; break; }
      depth--;
    }
  }

  if (start < end) return { start, end };
  return null;
}

/* ===================== RENDER ===================== */

function render() {
  const segs = getSegments(fullDoc);
  edit.value = segs[mode] || "";
  highlight();
}

function highlight() {
  const text = edit.value;
  const pos = edit.selectionStart;

  let block = null;
  if (mode !== "html") block = findBraceBlock(text, pos);

  if (!block) {
    view.textContent = text;
    msg.textContent = "No valid block";
    return;
  }

  msg.textContent = "";
  view.innerHTML =
    escape(text.slice(0, block.start)) +
    `${escape(text.slice(block.start, block.end))}` +
    escape(text.slice(block.end));
}

function escape(s) {
  return s.replace(/[&<>]/g, m =>
    ({ "&":"&", "<":"<", ">":">" }[m])
  );
}

/* ===================== EVENTS ===================== */

edit.addEventListener("input", () => {
  fullDoc = setSegment(fullDoc, mode, edit.value);
  highlight();
});

edit.addEventListener("keyup", highlight);
edit.addEventListener("click", highlight);

document.querySelectorAll("button[data-mode]").forEach(b => {
  b.onclick = () => {
    document.querySelectorAll("button").forEach(x => x.classList.remove("active"));
    b.classList.add("active");
    mode = b.dataset.mode;
    render();
  };
});

document.getElementById("replace").onclick = async () => {
  const text = edit.value;
  const pos = edit.selectionStart;
  const block = mode === "html" ? null : findBraceBlock(text, pos);

  if (!block) {
    msg.textContent = "Cannot replace: no block";
    return;
  }

  const clip = await navigator.clipboard.readText();
  edit.value = text.slice(0, block.start) + clip + text.slice(block.end);
  edit.dispatchEvent(new Event("input"));
};

/* ===================== INIT ===================== */
render();
