\n\n\n";

const edit = document.getElementById("edit");
const view = document.getElementById("view");
const msg  = document.getElementById("msg");

/* ================= REGEX (SAFE) ================= */
const RX_SCRIPT_ALL = new RegExp("", "gi");
const RX_STYLE_ALL  = new RegExp("", "gi");
const RX_SCRIPT_ONE = new RegExp("]*>([\\s\\S]*?)<\\/script>", "i");
const RX_STYLE_ONE  = new RegExp("]*>([\\s\\S]*?)<\\/style>", "i");

/* ================= SEGMENTS ================= */
function getSegments(doc) {
  const html = doc
    .replace(RX_SCRIPT_ALL, "")
    .replace(RX_STYLE_ALL, "");

  const jsMatch  = doc.match(RX_SCRIPT_ONE);
  const cssMatch = doc.match(RX_STYLE_ONE);

  return {
    html: html,
    js:  jsMatch  ? jsMatch[1]  : "",
    css: cssMatch ? cssMatch[1] : ""
  };
}

function setSegment(doc, mode, content) {
  if (mode === "html") {
    return doc
      .replace(RX_SCRIPT_ALL, "")
      .replace(RX_STYLE_ALL, "")
      .replace(/<\/head>/i, content + "\n");
  }

  if (mode === "js") {
    return doc.replace(
      RX_SCRIPT_ONE,
      ""
    );
  }

  if (mode === "css") {
    return doc.replace(
      RX_STYLE_ONE,
      ""
    );
  }

  return doc;
}

/* ================= BLOCK DETECTION ================= */
function findBraceBlock(text, pos) {
  let start = -1, depth = 0;

  for (let i = pos; i >= 0; i--) {
    if (text[i] === "}") depth++;
    if (text[i] === "{") {
      if (depth === 0) { start = i; break; }
      depth--;
    }
  }
  if (start === -1) return null;

  depth = 0;
  for (let i = start; i < text.length; i++) {
    if (text[i] === "{") depth++;
    if (text[i] === "}") {
      depth--;
      if (depth === 0) return { start, end: i + 1 };
    }
  }
  return null;
}

/* ================= RENDER ================= */
function render() {
  const segs = getSegments(fullDoc);
  edit.value = segs[mode] || "";
  highlight();
}

function highlight() {
  if (isIOS) return;

  const text = edit.value;
  const pos = edit.selectionStart;

  if (mode === "html") {
    view.textContent = text;
    msg.textContent = "";
    return;
  }

  const block = findBraceBlock(text, pos);
  if (!block) {
    view.textContent = text;
    msg.textContent = "No valid block";
    return;
  }

  msg.textContent = "";
  view.innerHTML =
    esc(text.slice(0, block.start)) +
    "" +
    esc(text.slice(block.start, block.end)) +
    "" +
    esc(text.slice(block.end));
}

function esc(s) {
  return s.replace(/[&<>]/g, m => (
    { "&":"&", "<":"<", ">":">" }[m]
  ));
}

/* ================= EVENTS ================= */
edit.addEventListener("input", () => {
  fullDoc = setSegment(fullDoc, mode, edit.value);
  highlight();
});

edit.addEventListener("keyup", highlight);
edit.addEventListener("click", highlight);

document.querySelectorAll("button[data-mode]").forEach(b => {
  b.onclick = () => {
    document.querySelectorAll("button").forEach(x => x.classList.remove("active"));
    b.classList.add("active");
    mode = b.dataset.mode;
    render();
  };
});

document.getElementById("replace").onclick = async () => {
  if (mode === "html") return;

  const text = edit.value;
  const pos = edit.selectionStart;
  const block = findBraceBlock(text, pos);

  if (!block) {
    msg.textContent = "Cannot replace block";
    return;
  }

  const clip = await navigator.clipboard.readText();
  edit.value = text.slice(0, block.start) + clip + text.slice(block.end);
  edit.dispatchEvent(new Event("input"));
};

/* ================= INIT ================= */
edit.value = "";     // disable Safari restore
render();
